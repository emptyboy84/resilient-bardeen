    const REDUCE_MOTION = typeof window !== 'undefined'
      && window.matchMedia
      && window.matchMedia('(prefers-reduced-motion: reduce)').matches;

    function startAutoplay() {
      stopAutoplay();
      if (REDUCE_MOTION) return;

      autoplay = setInterval(function() {
        const pool = pickCandidates();

        const changed =
          JSON.stringify(pool.map(function(ad) { return ad.id; })) !==
          JSON.stringify(currentPool.map(function(ad) { return ad.id; }));

        if (changed) {
          currentPool = pool.length ? pool : allAds.slice();
          idx = offsetIndex(currentPool);
          render(currentPool[idx]);
          rebuildDots(currentPool.length, idx);
          return;
        }

        if (Date.now() - lastUserActionAt < USER_PAUSE_MS) return;

        if (currentPool.length > 1) {
          idx = (idx + 1) % currentPool.length;
          render(currentPool[idx]);
          highlightDot(idx);
        }
      }, INTERVAL_MS);
    }

    function stopAutoplay() {
      if (autoplay) clearInterval(autoplay);
      autoplay = null;
    }

    function goTo(i, user) {
      if (!currentPool.length) return;

      idx = ((i % currentPool.length) + currentPool.length) % currentPool.length;
      render(currentPool[idx]);
      highlightDot(idx);

      if (user) lastUserActionAt = Date.now();
    }

    startAutoplay();
  })();

(function () {
    const topButton = document.getElementById("topButton");
    const bottomButton = document.getElementById("bottomButton");
    const reportButton = document.getElementById("companyReportButton");

    if (!topButton || !bottomButton) {
        return;
    }

    const detailMatched = (window.location.pathname || "").match(/^\/companyDetail\/([0-9]{10})$/i);
    const isCompanyDetail = !!detailMatched;
    const currentBno = detailMatched ? detailMatched[1] : "";

    topButton.dataset.bound = "Y";
    bottomButton.dataset.bound = "Y";

    if (reportButton) {
        reportButton.dataset.bound = "Y";

        if (!isCompanyDetail) {
            reportButton.remove();
        } else {
            reportButton.classList.add("show");
        }
    }

    function textOf(selector) {
        const el = document.querySelector(selector);
        return el && el.textContent ? el.textContent.replace(/\s+/g, " ").trim() : "";
    }

    function getCurrentCompanyName() {
        return textOf(".company-detail.companyname")
            || textOf(".search-list-group.companyname")
            || textOf(".coNm")
            || "";
    }

    function openCompanyDataReportPopup() {
        if (!isCompanyDetail) {
            return false;
        }

        const qs = new URLSearchParams();
        qs.set("bNo", currentBno);
        qs.set("coNm", getCurrentCompanyName());
        qs.set("pageUrl", window.location.href);
        qs.set("_ts", String(Date.now()));

        const src = "/mail/contact/report?" + qs.toString();

        if (window.MBZCommonPopup && typeof window.MBZCommonPopup.buildFramePopup === "function") {
            window.MBZCommonPopup.buildFramePopup(
                "mbzCompanyDataReportWrap",
                "데이터 오류·정정 요청",
                src,
                "920px",
                "min(650px, calc(100vh - 32px))"
            );
            return false;
        }

        $("<iframe id=\"iframeCompanyDataReportPop\" class=\"pop_iframe on\" src=\"" + src + "\" frameborder=\"0\" " +
            "style=\"left:0; top:0; width:100%; height:100vh; position:fixed; z-index:999999; border:0; backdrop-filter:blur(3px);\" />")
            .appendTo("body");

        if (window.MBZScrollLock) {
            window.MBZScrollLock.lock();
        } else {
            $("body").css("overflow-y", "hidden");
        }

        return false;
    }

    function syncPageMoveButtons() {
        const scrollTop = window.scrollY || document.documentElement.scrollTop || 0;
        const maxScroll = Math.max(
            0,
            document.documentElement.scrollHeight - window.innerHeight
        );

        if (scrollTop > 200) {
            topButton.classList.add("show");
        } else {
            topButton.classList.remove("show");
        }

        if ((maxScroll - scrollTop) > 200) {
            bottomButton.classList.add("show");
        } else {
            bottomButton.classList.remove("show");
        }

        if (reportButton && isCompanyDetail) {
            reportButton.classList.add("show");
        }
    }

    if (reportButton && isCompanyDetail) {
        reportButton.addEventListener("click", openCompanyDataReportPopup);
    }

    topButton.addEventListener("click", function () {
        window.scrollTo({
            top: 0,
            behavior: "smooth"
        });
    });

    bottomButton.addEventListener("click", function () {
        window.scrollTo({
            top: document.documentElement.scrollHeight,
            behavior: "smooth"
        });
    });

    window.addEventListener("scroll", syncPageMoveButtons, { passive: true });
    window.addEventListener("resize", syncPageMoveButtons);

    syncPageMoveButtons();
})();

/* R120: non-detail pages render this fragment with empty lists, then hydrate once via /right_panel.
 * The outer #rightPanel survives innerHTML replacement, so the started marker prevents loops. */
(function rpHydrateGlobalRightPanelOnce() {
    var root = document.getElementById('rightPanel');
    if (!root || root.hasAttribute('data-deferred-right-panel')) return;
    if (root.getAttribute('data-mbz-right-panel-hydration-started') === '1') return;

    var attempts = 0;
    function start() {
        if (root.getAttribute('data-mbz-right-panel-hydration-started') === '1') return;
        if (typeof window.refreshRightPanel !== 'function') {
            attempts += 1;
            if (attempts <= 20) window.setTimeout(start, 100);
            return;
        }
        root.setAttribute('data-mbz-right-panel-hydration-started', '1');
        window.refreshRightPanel({ telemetry: false });
    }
    start();
})();

(function rpBootRightPanelNow() {
    var rightPanel = document.getElementById('rightPanel');

    if (!rightPanel) {
        return;
    }

    var realtimeSearchTab = rightPanel.querySelector('#realtimeSearchTab');
    var historySearchTab = rightPanel.querySelector('#historySearchTab');

    if (!realtimeSearchTab && !historySearchTab) {
        return;
    }

    var rightPanelActiveTabButton = null;

    try {
        rightPanelActiveTabButton = JSON.parse(localStorage.getItem('rightPanelActiveTabButton'));
    } catch (e) {
        rightPanelActiveTabButton = null;
        localStorage.removeItem('rightPanelActiveTabButton');
    }

    if (rightPanelActiveTabButton
            && rightPanelActiveTabButton.id
            && rightPanel.querySelector('#' + rightPanelActiveTabButton.id)) {
        switchTab(rightPanelActiveTabButton.id);
    } else if (realtimeSearchTab) {
        switchTab('realtimeSearchTab');
    } else if (historySearchTab) {
        switchTab('historySearchTab');
    }

    if (realtimeSearchTab) {
        realtimeSearchTab.addEventListener('click', function() {
            localStorage.setItem('rightPanelActiveTabButton', JSON.stringify({
                name: 'activeTabButton',
                id: 'realtimeSearchTab'
            }));
            switchTab('realtimeSearchTab');
        });
    }

    if (historySearchTab) {
        historySearchTab.addEventListener('click', function() {
            localStorage.setItem('rightPanelActiveTabButton', JSON.stringify({
                name: 'activeTabButton',
                id: 'historySearchTab'
            }));
            switchTab('historySearchTab');
        });
    }

    if (typeof rpHydrateRightPanelFinanceCharts === 'function') {
        /* R4: start entitlement-gated charts synchronously while the shell XHR callback is hot. */
        rpHydrateRightPanelFinanceCharts();
    }

    if (typeof rpInitRightPanelCharts === 'function') {
        setTimeout(rpInitRightPanelCharts, 0);

        var rpResizeTimer;
        $(window)
            .off('resize.mbzRightPanelCharts')
            .on('resize.mbzRightPanelCharts', function() {

    }

function switchTab(tabName) {
    var rightPanel = document.getElementById('rightPanel');

    if (!rightPanel) {
        return false;
    }

    var targetId = tabName === 'historySearchTab' ? 'historySearchTab' : 'realtimeSearchTab';
    var activeButton = rightPanel.querySelector('#' + targetId);

    if (!activeButton) {
        return false;
    }

    rightPanel.querySelectorAll('.tab-button').forEach(function(button) {
        button.classList.remove('active');
    });

    rightPanel.querySelectorAll('.item').forEach(function(item) {
        item.classList.remove('active');
    });

    activeButton.classList.add('active');

    var realtimeSearchListY = rightPanel.querySelector('#realtimeSearchListY');
    var realtimeSearchListN = rightPanel.querySelector('#realtimeSearchListN');

    if (targetId === 'historySearchTab') {
        rightPanel.querySelectorAll('.item.memb-his-search').forEach(function(item) {
            item.classList.add('active');
        });

        if (realtimeSearchListY) {
            realtimeSearchListY.style.display = '';
        }

        if (realtimeSearchListN) {
            realtimeSearchListN.style.display = 'none';
        }
    } else {
        rightPanel.querySelectorAll('.item:not(.memb-his-search)').forEach(function(item) {
            item.classList.add('active');
        });

        if (realtimeSearchListY) {
            realtimeSearchListY.style.display = 'none';
        }

        if (realtimeSearchListN) {
            realtimeSearchListN.style.display = '';
        }
    }

    if (typeof defineWidth === 'function') {
        defineWidth();
    }

    if (typeof rpHydrateRightPanelFinanceCharts === 'function') {
        /* R4: start entitlement-gated charts synchronously while the shell XHR callback is hot. */
        rpHydrateRightPanelFinanceCharts();
    }

    if (typeof rpInitRightPanelCharts === 'function') {
        setTimeout(rpInitRightPanelCharts, 0);
    }

    return false;
}

/* 우측패널 미니차트 렌더 유틸 */
function rpHasMiniChartPayload(el) {
    if (!el) return false;
    const raw = (el.getAttribute('data-sls') || '').trim();
    return raw !== '' && raw !== 'null' && raw !== '[]';
}

function rpHydrateRightPanelFinanceCharts() {
    const rightPanel = document.getElementById('rightPanel');
    if (!rightPanel || typeof window.jQuery === 'undefined') return;

    const pending = Array.prototype.filter.call(
        rightPanel.querySelectorAll('.panel .inner .item.active .rp-finance-paid[data-bno]'),
        function(el) {
            return el.dataset.chartRequested !== '1' && !rpHasMiniChartPayload(el);
        }
    );
    if (!pending.length) return;

    const bnos = [];
    pending.forEach(function(el) {
        const bno = (el.getAttribute('data-bno') || '').trim();
        if (/^\d{10}$/.test(bno) && bnos.indexOf(bno) < 0) {
            bnos.push(bno);
        }
        el.dataset.chartRequested = '1';
    });
    if (!bnos.length) return;

    $.ajax({
        url: '/right_panel/finance_charts',
        method: 'GET',
        traditional: true,
        dataType: 'json',
        data: { bno: bnos },
        headers: {
            'X-Requested-With': 'XMLHttpRequest',
            'X-MarketBZ-Right-Panel-Intent': 'embedded-auto'
        }
    }).done(function(rows) {
        const byBno = {};
        (Array.isArray(rows) ? rows : []).forEach(function(row) {
            if (row && row.bno) byBno[String(row.bno).trim()] = row;
        });

        pending.forEach(function(el) {
            const bno = (el.getAttribute('data-bno') || '').trim();
            const row = byBno[bno];
            if (row && row.slsAmtChartArr) {
                el.setAttribute('data-sls', row.slsAmtChartArr);
                el.setAttribute('data-colors', row.slsAmtChartColorArr || '[]');
                el.dataset.rendered = '0';
                el.classList.remove('hidden');
            } else {
                el.setAttribute('data-sls', '[]');
                el.setAttribute('data-colors', '[]');
                el.dataset.rendered = '1';
                el.classList.add('hidden');
            }
        });

        rpInitRightPanelCharts();
    }).fail(function() {
        pending.forEach(function(el) {
            delete el.dataset.chartRequested;
        });
    });
}

function rpInitRightPanelCharts() {
    const $charts = $('.right-panel .panel .inner .item.active .sls_chart').filter(function() {
        return rpHasMiniChartPayload(this);
    });
    if (!$charts.length) return;

    const useIO = 'IntersectionObserver' in window;

    if (useIO) {
        const io = new IntersectionObserver(entries => {
            entries.forEach(entry => {
                if (entry.isIntersecting) {
                    rpDrawMiniChart(entry.target);
                    io.unobserve(entry.target);
                }
            });
        });

        $charts.each(function() {
            if (this && this.dataset.rendered !== '1') {
                io.observe(this);
            }
        });
    } else {
        $charts.each(function() {
            rpDrawMiniChart(this);
        });
    }
}

function rpDrawMiniChart(el) {
    if (!el || el.dataset.rendered === '1') return;

    // company-list의 drawMiniChart가 있으면 그대로 사용
    if (typeof window.drawMiniChart === 'function') {
        try {
            window.drawMiniChart(el);
            el.dataset.rendered = '1';
            return;
