```javascript
(function autoNextDreamCampus() {
    console.log("%c[DreamCampus 자동 다음 넘김 활성화됨]", "color: #00ff00; font-size: 16px; font-weight: bold;");
    
    setInterval(() => {
        try {
            // 1. 차시 완료 팝업 ("다음 차시 학습 하기") 자동 클릭
            const nextTimeBtn = document.querySelector('.jconfirm-buttons .btn-purple, .jconfirm-buttons button');
            if (nextTimeBtn && (nextTimeBtn.innerText.includes('다음 차시') || nextTimeBtn.innerText.includes('확인'))) {
                console.log('[자동 다음] 다음 차시 학습 진행');
                nextTimeBtn.click();
                return;
            }

            // 2. iframe 내부 영상(video/audio) 완료 감지
            const iframe = document.getElementById('iframe');
            if (iframe && iframe.contentWindow) {
                const iframeDoc = iframe.contentDocument || iframe.contentWindow.document;
                if (iframeDoc) {
                    const media = iframeDoc.querySelector('video, audio');
                    if (media) {
                        // 정지 상태인 경우 자동 재생
                        if (media.paused && media.currentTime === 0) {
                            media.play().catch(() => {});
                        }
                        // 영상 종료 시 다음 페이지 버튼 클릭
                        if (media.ended || (media.duration > 0 && media.currentTime >= media.duration - 0.5)) {
                            const nextBtn = document.getElementById('page_next');
                            if (nextBtn) {
                                console.log('[자동 다음] 영상 완료 -> 다음 페이지로 이동');
                                nextBtn.click();
                            }
                        }
                        return;
                    }
                }
            }

            // 3. 영상이 없는 일반 페이지의 경우 시간 충족(arrow 활성화) 시 자동 클릭
            const nextBtn = document.getElementById('page_next');
            if (nextBtn && nextBtn.classList.contains('arrow')) {
                console.log('[자동 다음] 페이지 학습 완료 -> 다음 클릭');
                nextBtn.click();
            }
        } catch (e) {
            // 에러 무시
        }
    }, 2000);
})();
```

- **영상 종료 자동 감지**: 동영상이 끝나면 2초 내에 하단의 **[다음 페이지]** 버튼을 자동으로 클릭합니다.
- **차시 완료 자동 이동**: 1개 차시가 모두 끝나 팝업("다음 차시 학습 하기")이 뜨면 자동으로 승인하여 다음 차시로 넘어갑니다.
- **시작 시 자동 재생**: 페이지가 넘어가면서 영상이 멈춰있으면 자동으로 재생을 시작합니다.
